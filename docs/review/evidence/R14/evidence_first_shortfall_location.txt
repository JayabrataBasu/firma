--- how attend() reduces the three to one:
    pub fn attend(h: f64, shortfalls: [f64; 3], h_crit: f64) -> Focus {
        if h < h_crit {
            Focus::Survival
        } else {
            let mut best = 0usize;
            for j in 1..3 {
                if shortfalls[j] > shortfalls[best] {
                    best = j;
                }
            }
            if shortfalls[best] > 0.0 {
                Focus::Goal((best + 1) as u8)
            } else {
                Focus::None
            }
        }
    }
